/**
 * Formatea un importe en pesos argentinos.
 * @param {number} precio
 * @returns {string}
 */
const formatearPrecio = (precio) =>
  new Intl.NumberFormat("es-AR", { style: "currency", currency: "ARS" }).format(precio);

/**
 * Obtiene la imagen del material de clase.
 * @param {object} producto
 * @returns {string}
 */
const imagenProducto = (producto) =>
  "https://ucc-tallerdesarrolloweb.github.io/filminas/images/ejercicios/" + producto.imagen;

/**
 * Filtra y ordena una copia, sin modificar el catálogo.
 * @param {object} filtros
 * @returns {object[]}
 */
const seleccionarProductos = (filtros) => {
  const lista = productos.filter((producto) =>
    producto.nombre.toLowerCase().includes(filtros.texto.toLowerCase().trim()) &&
    producto.precio >= filtros.minimo && producto.precio <= filtros.maximo &&
    (filtros.marca === "Todas" || producto.marca === filtros.marca) &&
    (filtros.categorias.length === 0 || filtros.categorias.includes(producto.categoria))
  );
  lista.sort((a, b) => {
    if (filtros.orden === "precio-desc") return b.precio - a.precio;
    if (filtros.orden === "precio-asc") return a.precio - b.precio;
    const comparacion = a.nombre.toLowerCase() < b.nombre.toLowerCase() ? -1 :
      a.nombre.toLowerCase() > b.nombre.toLowerCase() ? 1 : 0;
    return filtros.orden === "nombre-desc" ? -comparacion : comparacion;
  });
  return lista;
};

/**

const cargarProductos = (lista) => {

      <p class="precio">${formatearPrecio(producto.precio)}</p>
      <button type="button" onclick="mostrarDetalle(${producto.id})">Ver detalle de ${producto.nombre}</button>
      <label for="cantidad-${producto.id}">Cantidad de ${producto.nombre}</label>
      <input id="cantidad-${producto.id}" type="number" min="1" max="99" step="1" value="1">
      <button type="button" onclick="agregarAlCarrito(${producto.id})">Agregar al carrito</button>
    </article>`
  ).join("");
};

/**
 * Lee los controles y actualiza el catálogo; valida el rango de precios.
 * @returns {void}

const filtrarProductos = () => {
  const minimo = Number(document.getElementById("precio-min").value);
  const maximoTexto = document.getElementById("precio-max").value;
  const maximo = maximoTexto === "" ? Infinity : Number(maximoTexto);
  if (!Number.isFinite(minimo) || minimo < 0 || maximo < minimo || Number.isNaN(maximo)) {
    alert("Revisá los precios: el mínimo no puede superar al máximo ni ser negativo.");
    return;
  }
  const categorias = [];
  if (document.getElementById("protectores").checked) categorias.push("Protectores");
  if (document.getElementById("entrenamiento").checked) categorias.push("Entrenamiento");
  if (document.getElementById("dobok").checked) categorias.push("Dobok");
  const lista = seleccionarProductos({
    texto: document.getElementById("search").value,
    minimo, maximo, categorias,
    marca: document.getElementById("marca").value,
    orden: document.getElementById("order").value
  });
  cargarProductos(lista);
  document.getElementById("mensaje-tienda").textContent = lista.length === 0 ?
    "No hay productos que coincidan con los filtros." : lista.length + " productos encontrados.";
};

/**
 * Comprueba el límite de unidades por producto.
 * @param {number} cantidad
 * @returns {boolean}
 */
const cantidadValida = (cantidad) => Number.isInteger(cantidad) && cantidad >= 1 && cantidad <= 99;

/**
 * Recupera únicamente registros válidos del navegador.
 * @returns {object[]}
 */
const leerCarrito = () => {
  try {
    const guardado = JSON.parse(localStorage.getItem("carrito"));
    if (!Array.isArray(guardado)) return [];
    const carrito = [];
    guardado.forEach((item) => {
      if (!item || !productos.some((producto) => producto.id === item.id) ||
          !cantidadValida(item.cantidad)) return;
      const existente = carrito.find((producto) => producto.id === item.id);
      if (!existente) carrito.push({ id: item.id, cantidad: item.cantidad });
    });
    return carrito;
  } catch (error) {
    return [];
  }
};

/**
 * Guarda el carrito o elimina su clave cuando queda vacío.
 * @param {object[]} carrito
 * @returns {boolean}
 */
const guardarCarrito = (carrito) => {
  try {
    if (carrito.length === 0) localStorage.removeItem("carrito");
    else localStorage.setItem("carrito", JSON.stringify(carrito));
    return true;
  } catch (error) {
    alert("No se pudo guardar el carrito. Revisá los permisos de almacenamiento del navegador.");
    return false;
  }
};

/**
 * Calcula el total usando los precios del catálogo.
 * @param {object[]} carrito
 * @returns {number}
 */
const calcularTotal = (carrito) => carrito.reduce((total, item) =>
  total + productos.find((producto) => producto.id === item.id).precio * item.cantidad, 0);

const contarProductos = () => {
  document.getElementById("cant-prod").textContent =
    leerCarrito().reduce((total, item) => total + item.cantidad, 0);

/**
 * Agrega unidades sin duplicar la entrada del producto.
 * @param {number} id
 * @returns {void}
 */
const agregarAlCarrito = (id) => {
  if (!productos.some((producto) => producto.id === id)) return;
  const campo = document.getElementById("cantidad-" + id);
  const cantidad = Number(campo.value);
  const carrito = leerCarrito();
  const existente = carrito.find((item) => item.id === id);
  if (!cantidadValida(cantidad) || (existente && !cantidadValida(existente.cantidad + cantidad))) {
    alert("Ingresá una cantidad entera entre 1 y 99. El acumulado tampoco puede superar 99.");
    campo.value = "";
    campo.focus();
    return;
  }
  if (existente) existente.cantidad += cantidad;
  else carrito.push({ id, cantidad });
  if (!guardarCarrito(carrito)) return;
  contarProductos();
  document.getElementById("mensaje-tienda").textContent = "Producto agregado al carrito.";
};

/**
 * Actualiza total, contador y disponibilidad del botón de vaciado.
 * @param {object[]} carrito
 * @returns {void}
 */
const actualizarTotales = (carrito) => {
  contarProductos();
  document.getElementById("total-carrito").textContent = formatearPrecio(calcularTotal(carrito));
  document.getElementById("vaciar-carrito").disabled = carrito.length === 0;
};

/**
 * Renderiza los productos guardados, o informa que el carrito está vacío.
 * @returns {void}
 */
const mostrarCarrito = () => {
  const carrito = leerCarrito();
  document.getElementById("mostrar-carrito").innerHTML = carrito.length === 0 ?
    "<p>Tu carrito está vacío. Podés agregar productos desde el catálogo.</p>" :
    carrito.map((item) => {
      const producto = productos.find((producto) => producto.id === item.id);
      return `<article class="tarjeta">
        <h2>${producto.nombre}</h2>
        <p>Precio por unidad: ${formatearPrecio(producto.precio)}</p>
        <label for="unidades-${item.id}">Cantidad de ${producto.nombre}</label>
        <input id="unidades-${item.id}" type="number" min="1" max="99" step="1"
          value="${item.cantidad}" onchange="cambiarCantidad(${item.id})">
        <p>Subtotal: <span id="subtotal-${item.id}">${formatearPrecio(producto.precio * item.cantidad)}</span></p>
        <button type="button" onclick="eliminarProducto(${item.id})">Eliminar ${producto.nombre}</button>
      </article>`;
    }).join("");
  actualizarTotales(carrito);
};

/**
 * Cambia unidades sin reconstruir el campo enfocado.
 * @param {number} id
 * @returns {void}
 */
const cambiarCantidad = (id) => {
  const carrito = leerCarrito();
  const item = carrito.find((producto) => producto.id === id);
  if (!item) return;
  const campo = document.getElementById("unidades-" + id);
  const cantidad = Number(campo.value);
  if (!cantidadValida(cantidad)) {
    alert("Ingresá una cantidad entera entre 1 y 99.");
    campo.value = item.cantidad;
    campo.focus();
    return;
  }
  item.cantidad = cantidad;
  if (!guardarCarrito(carrito)) return;
  const producto = productos.find((producto) => producto.id === id);
  document.getElementById("subtotal-" + id).textContent = formatearPrecio(producto.precio * cantidad);
  actualizarTotales(carrito);
  document.getElementById("mensaje-tienda").textContent = "Cantidad actualizada.";
};

/**
 * Elimina una entrada completa con splice.
 * @param {number} id
 * @returns {void}
 */
const eliminarProducto = (id) => {
  const carrito = leerCarrito();
  const indice = carrito.findIndex((item) => item.id === id);
  if (indice === -1) return;
  carrito.splice(indice, 1);
  if (!guardarCarrito(carrito)) return;
  mostrarCarrito();
  document.getElementById("mensaje-tienda").textContent = "Producto eliminado.";
};

/**
 * Vacía el almacenamiento y actualiza la vista.
 * @returns {void}
 */
const vaciarCarrito = () => {
  if (!guardarCarrito([])) return;
  mostrarCarrito();
  document.getElementById("mensaje-tienda").textContent = "Carrito vaciado.";
};
